import { useState, useEffect } from 'react';
import axios from 'axios';
import { Card, CardContent, Typography, Chip } from '@mui/material';

export default function AIInsights() {
    const [insights, setInsights] = useState([]);
    const [loading, setLoading] = useState(true);
    const [countdown, setCountdown] = useState('');

    useEffect(() => {
        fetchInsights();
    }, []);

    useEffect(() => {
        const tick = () => {
            const now = new Date();
            const totalSecs = (15 - (now.getMinutes() % 15)) * 60 - now.getSeconds();
            const m = Math.floor(totalSecs / 60) % 15;
            const s = totalSecs % 60;
            setCountdown(`${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`);
        };
        tick();
        const id = setInterval(tick, 1000);
        return () => clearInterval(id);
    }, []);

    const fetchInsights = async () => {
        try {
            const res = await axios.get('http://localhost:8000/errors/insights');
            setInsights(res.data);
        } catch (err) {
            console.error(err);
        }
        setLoading(false);
    };

    if (loading) return <div className="text-gray">Loading AI Insights...</div>;

    return (
        <div className="space-y-6">
            <div className="flex-between">
                <h2 className="page-title gradient-text">
                    AI-Powered Error Insights
                </h2>
                <div style={{ display: 'flex', alignItems: 'center', gap: '12px' }}>
                    <div style={{ display: 'flex', alignItems: 'center', gap: '6px', backgroundColor: 'var(--bg-darker)', border: '1px solid var(--border-color)', borderRadius: '8px', padding: '6px 12px', fontSize: '0.85rem' }}>
                        <span style={{ width: '8px', height: '8px', borderRadius: '50%', backgroundColor: '#22c55e', display: 'inline-block', animation: 'pulse 2s infinite' }}></span>
                        <span style={{ color: '#94a3b8' }}>Next update in</span>
                        <span style={{ color: '#22c55e', fontWeight: 'bold', fontFamily: 'monospace' }}>{countdown}</span>
                    </div>
                    <Chip label="Live Analysis" color="success" variant="outlined" />
                </div>
            </div>

            <div className="insights-grid">
                {insights.map((insight: any) => (
                    <Card key={insight._id} sx={{ backgroundColor: 'var(--bg-darker)', backgroundImage: 'none', border: '1px solid var(--border-color)', borderRadius: 2, boxShadow: '0 1px 3px 0 rgba(0, 0, 0, 0.1)' }}>
                        <CardContent>
                            <div className="ai-card-header">
                                <Chip
                                    label={insight.type.replace('_', ' ').toUpperCase()}
                                    size="small"
                                    color={insight.type === 'anomaly' ? 'error' : 'secondary'}
                                    sx={{ fontWeight: 'bold' }}
                                />
                                <span className="ai-date">
                                    {new Date(insight.timestamp).toLocaleString()}
                                </span>
                            </div>

                            {insight.type === 'summary' && (
                                <Typography variant="body1" sx={{ color: '#334155', fontStyle: 'italic', borderLeft: '4px solid #4f46e5', pl: 2 }}>
                                    "{insight.content}"
                                </Typography>
                            )}

                            {insight.type === 'anomaly' && (
                                <div className="anomaly-alert">
                                    <Typography variant="subtitle1" sx={{ color: '#b91c1c', fontWeight: 'bold' }}>
                                        🚨 Warning
                                    </Typography>
                                    <Typography variant="body2" sx={{ color: '#991b1b', mt: 1 }}>
                                        {insight.message}
                                    </Typography>
                                </div>
                            )}

                            {insight.type === 'clustering_and_rca' && (
                                <div style={{ display: 'flex', flexDirection: 'column', gap: '16px' }}>
                                    <Typography variant="h6" sx={{ color: '#4f46e5', fontSize: '1.1rem' }}>
                                        Root Cause Analysis
                                    </Typography>
                                    <Typography variant="body2" sx={{ color: '#334155' }}>
                                        <strong>Probable Cause:</strong> {insight.root_cause_analysis?.probable_root_cause}
                                    </Typography>
                                    <Typography variant="body2" sx={{ color: '#334155' }}>
                                        <strong>Impacted:</strong> {insight.root_cause_analysis?.impacted_services?.join(', ')}
                                    </Typography>

                                    <div style={{ marginTop: '16px', paddingTop: '16px', borderTop: '1px solid var(--border-color)' }}>
                                        <Typography variant="subtitle2" sx={{ color: '#64748b', mb: 2 }}>
                                            Identified Clusters ({insight.clusters?.length || 0})
                                        </Typography>
                                        <div style={{ display: 'flex', flexDirection: 'column', gap: '8px' }}>
                                            {insight.clusters?.map((c: any, i: number) => (
                                                <div key={i} className="cluster-item">
                                                    <div className="cluster-meta">Cluster {c.cluster_id} • Size: {c.size}</div>
                                                    <div className="cluster-msg" title={c.sample_messages?.[0]}>{c.sample_messages?.[0]}</div>
                                                </div>
                                            ))}
                                        </div>
                                    </div>
                                </div>
                            )}
                        </CardContent>
                    </Card>
                ))}
                {insights.length === 0 && (
                    <div className="empty-insights">
                        No AI insights generated yet. AI models are processing logs.
                    </div>
                )}
            </div>
        </div>
    );
}
